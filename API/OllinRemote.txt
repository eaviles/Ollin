enum RemoteControlKind: String, Decodable, Encodable, Equatable, Hashable, RawRepresentable, Sendable
  case color
  case insets
  case menu
  case range
  case rect
  case slider
  case stepper
  case swatches
  case text
  case toggle
  case vector
  case vector3
enum RemoteHTTP
  static func acceptKey(for: String) -> String
  static func isWebSocketUpgrade(_: RemoteHTTP.Request) -> Bool
  static func notFound() -> Foundation.Data
  static func pageResponse(html: Foundation.Data) -> Foundation.Data
  static func parseHead(_: [UInt8]) -> (request: RemoteHTTP.Request, consumed: Int)?
  static func upgradeResponse(accept: String) -> Foundation.Data
  struct Request: Equatable, Sendable
    var headers: [String : String] { get set }
    var method: String { get set }
    var path: String { get set }
enum RemoteMessageKind: String, Decodable, Encodable, Equatable, Hashable, RawRepresentable, Sendable
  case hello
  case set
  case stats
  case update
enum RemoteWire
  static func apply(_: ParamStored, to: ParamHandle)
  static func descriptor(for: ParamHandle) -> RemoteParamDescriptor
  static func snapshotValue(of: ParamHandle) -> ParamStored
enum WebSocketFraming
  enum Opcode: UInt8, Equatable, Hashable, RawRepresentable, Sendable
    case binary
    case close
    case continuation
    case ping
    case pong
    case text
  static func decode(buffer: [UInt8]) -> [WebSocketFraming.Frame]?
  static func encode(_: WebSocketFraming.Opcode, payload: [UInt8]) -> Foundation.Data
  static func encodeText(_: String) -> Foundation.Data
  struct Frame: Equatable, Sendable
    init(fin: Bool, opcode: WebSocketFraming.Opcode, payload: [UInt8])
    var fin: Bool { get set }
    var opcode: WebSocketFraming.Opcode { get set }
    var payload: [UInt8] { get set }

  var name: String { get set }
  var options: [String]? { get set }
  var step: Double? { get set }
  var upper: Double? { get set }
  var value: ParamStored { get set }
  var wLower: Double? { get set }
  var wUpper: Double? { get set }
  var xLower: Double? { get set }
  var xUpper: Double? { get set }
  var yLower: Double? { get set }
  var yUpper: Double? { get set }
  var zLower: Double? { get set }
  var zUpper: Double? { get set }
